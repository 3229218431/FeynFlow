```plantuml
@startuml 静电场与恒定电流场对偶关系
!theme plain
skinparam backgroundColor #FEFEFE

package "静电场 (Electrostatic) §2" as ES #LightYellow {
  class "∇×E = 0\n无旋场" as es_curl
  class "∇·D = ρ\n高斯定律" as es_div
  class "D = εE\n本构关系" as es_const
  class "E = -∇φ\n电位梯度" as es_pot
  class "∮D·dS = Q\n高斯积分" as es_int
  class "Q = CV\n电容定义" as es_C
}

package "恒定电流场 (Steady Current) §4" as SC #LightBlue {
  class "∇×E = 0\n无旋(相同)" as sc_curl
  class "∇·J = 0\n电流连续(ρ=0)" as sc_div
  class "J = σE\n欧姆定律" as sc_const
  class "E = -∇φ\n电位梯度(相同)" as sc_pot
  class "∮J·dS = 0\nKCL形式" as sc_int
  class "G = I/U\n电导定义" as sc_G
}

package "对偶量替换表" as DUAL #LightGreen {
  class "D↔J\n电通密度↔电流密度" as dual1
  class "ε↔σ\n介电常数↔电导率" as dual2
  class "Q↔I\n电荷↔电流" as dual3
  class "C↔G\n电容↔电导" as dual4
  class "ρ↔0\n(无源区)" as dual5
  class "E电位差↔U电压" as dual6
}

es_curl -[hidden]right-> es_div
es_div -[hidden]right-> es_const
es_pot -[hidden]down-> es_int
es_int -[hidden]down-> es_C

sc_curl -[hidden]right-> sc_div
sc_div -[hidden]right-> sc_const
sc_pot -[hidden]down-> sc_int
sc_int -[hidden]down-> sc_G

es_curl <..> sc_curl : 完全相同
es_div <..> sc_div : 趋于0(无源区)
es_const <..> sc_const : ε→σ替换
es_int <..> sc_int : D→J替换
es_C <..> sc_G : C→G替换

DUAL -up-> es_curl
DUAL -up-> sc_curl

note bottom of es_div
  静电场中有自由电荷ρ
  恒定电流场中∇·J=0
  (但界面处有面电荷)
end note

note bottom of sc_div
  ∇·J=0 稳恒条件
  但时变场中∇·J=-∂ρ/∂t
end note

note right of DUAL
  对偶替换法:
  已知静电场解 → 替换ε→σ
  → 直接得到恒定电流场解
  例: 球壳电阻R=1/(2πσ)(1/a-1/b)
  对应: 同心球电容C=4πε/(1/a-1/b)
end note

@enduml
```
